function $(id) { return document.getElementById(id); }
function toast(msg, kind = 'info') {
  const t = $('toast');
  t.className = 'toast show ' + kind;
  t.textContent = msg;
  clearTimeout(window._toastT);
  window._toastT = setTimeout(() => t.classList.remove('show'), 3200);
}
function callGS(fn, ...args) {
  return new Promise((resolve, reject) => {
    google.script.run.withSuccessHandler(resolve).withFailureHandler(err => {
      console.error('GS error:', err);
      reject(err);
    })[fn](...args);
  });
}
function fmtNum(n) { return (Number(n) || 0).toLocaleString(); }
function fmtSigned(n) { const v = Number(n) || 0; return (v > 0 ? '+' : '') + v.toLocaleString(); }
function esc(s) { return String(s == null ? '' : s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }
function todayISO() { return new Date().toISOString().slice(0, 10); }
function debounce(fn, ms) { let t; return (...a) => { clearTimeout(t); t = setTimeout(() => fn(...a), ms); }; }
/**
 * Rewrite a <datalist>'s options only if the content actually changed.
 * Browsers close an open native autocomplete popup the instant the backing
 * <datalist> is mutated — so re-writing it with identical content on every
 * keystroke (e.g. while an auto-refresh handler re-fetches the same list)
 * makes suggestions flash and immediately disappear while typing.
 */
function setDatalistHtml(dl, html) {
  if (!dl) return;
  if (dl._lastHtml === html) return;
  dl._lastHtml = html;
  dl.innerHTML = html;
}

function rowClass(status) {
  if (status === 'EXCESS' || status === 'OVER SCANNED') return 'diff-excess';
  if (status === 'SHORT' || status === 'UNDER SCANNED') return 'diff-short';
  if (status === 'MISSING PHYSICAL' || status === 'MISSING BOOK' ||
      status === 'SCAN WITHOUT BILL' || status === 'BILL WITHOUT SCAN') return 'diff-miss';
  if (status === 'PURCHASE PENDING' || status === 'CUSTOMER RETURN') return 'diff-short';
  return '';
}
function statusPill(s) {
  const map = {
    'MATCH': 'ok',
    'EXCESS': 'info', 'SHORT': 'hot',
    'OVER SCANNED': 'info', 'UNDER SCANNED': 'hot',
    'MISSING PHYSICAL': 'warn', 'MISSING BOOK': 'warn',
    'SCAN WITHOUT BILL': 'warn', 'BILL WITHOUT SCAN': 'warn',
    'PURCHASE PENDING': 'hot', 'CUSTOMER RETURN': 'info',
    'NOT ORDERED': 'hot',
    'Pending': 'hot', 'Partial': 'warn', 'Completed': 'ok',
  };
  return `<span class="pill ${map[s] || 'info'}">${s}</span>`;
}

function populateDatalists() {

/* --------- custom customer/billing autocomplete ---------
 * The native <datalist> approach was unreliable in this embedded webapp: the
 * suggestion popup flashed and closed instantly, and picking a name didn't
 * reliably populate dependent data. This is a small self-contained dropdown
 * driven entirely by our own JS/state, so it always shows and a click/Enter
 * always fires the same events a real datalist selection would.
 */
function uniqueCustomerNames() {
  const seen = new Set(); const out = [];
  (state.masters.customers || []).forEach(c => {
    const name = String(c['Customer Name'] || '').trim();
    if (!name) return;
    const key = name.toUpperCase();
    if (seen.has(key)) return;
    seen.add(key); out.push(name);
  });
  return out.sort((a, b) => a.localeCompare(b));
}
function uniqueBillingNames() {
  const seen = new Set(); const out = [];
  (state.masters.customers || []).forEach(c => {
    const name = String(c['Billing Name'] || '').trim();
    if (!name) return;
    const key = name.toUpperCase();
    if (seen.has(key)) return;
    seen.add(key); out.push(name);
  });
  return out.sort((a, b) => a.localeCompare(b));
}
function billingForCustomer(customerName) {
  const key = String(customerName || '').trim().toUpperCase();
  if (!key) return '';
  const row = (state.masters.customers || []).find(c => String(c['Customer Name'] || '').trim().toUpperCase() === key);
  return row ? String(row['Billing Name'] || '') : '';
}

const MAX_AUTOCOMPLETE_ITEMS = 50;
const _acCloseFns = [];
document.addEventListener('click', e => { _acCloseFns.forEach(fn => fn(e.target)); });

function setupAutocomplete(inputId, getOptions, onSelect) {
  const input = $(inputId);
  if (!input || input._acWired) return;
  input._acWired = true;
  input.removeAttribute('list');
  input.setAttribute('autocomplete', 'off');

  const wrap = document.createElement('div');
  wrap.className = 'ac-wrap';
  input.parentNode.insertBefore(wrap, input);
  wrap.appendChild(input);
  const menu = document.createElement('div');
  menu.className = 'ac-menu hidden';
  wrap.appendChild(menu);

  let items = [];
  let activeIdx = -1;

  function highlight() {
    [...menu.children].forEach((el, i) => el.classList.toggle('active', i === activeIdx));
    const el = menu.children[activeIdx];
    if (el) el.scrollIntoView({ block: 'nearest' });
  }
  function close() { menu.classList.add('hidden'); activeIdx = -1; }
  function render() {
    const q = input.value.trim().toLowerCase();
    const all = getOptions();
    items = (q ? all.filter(c => c.toLowerCase().includes(q)) : all).slice(0, MAX_AUTOCOMPLETE_ITEMS);
    activeIdx = -1;
    if (!items.length) { close(); menu.innerHTML = ''; return; }
    menu.innerHTML = items.map((c, i) => `<div class="ac-item" data-i="${i}">${esc(c)}</div>`).join('');
    menu.classList.remove('hidden');
  }
  function pick(name) {
    input.value = name;
    input.dispatchEvent(new Event('input', { bubbles: true }));
    input.dispatchEvent(new Event('change', { bubbles: true }));
    close();
    if (onSelect) onSelect(name);
  }

  input.addEventListener('focus', render);
  input.addEventListener('input', render);
  input.addEventListener('keydown', e => {
    if (menu.classList.contains('hidden')) return;
    if (e.key === 'ArrowDown') {
      e.preventDefault();
      activeIdx = Math.min(activeIdx + 1, items.length - 1);
      highlight();
    } else if (e.key === 'ArrowUp') {
      e.preventDefault();
      activeIdx = Math.max(activeIdx - 1, 0);
      highlight();
    } else if (e.key === 'Enter') {
      if (activeIdx >= 0) { e.preventDefault(); pick(items[activeIdx]); }
    } else if (e.key === 'Escape') {
      close();
    }
  });
  menu.addEventListener('mousedown', e => {
    const el = e.target.closest('.ac-item');
    if (!el) return;
    e.preventDefault();
    pick(items[Number(el.dataset.i)]);
  });
  _acCloseFns.push(target => { if (!wrap.contains(target)) close(); });
}

function setupCustomerAutocompletes() {
  setupAutocomplete('ord-cust', uniqueCustomerNames, name => {
    const billing = billingForCustomer(name);
    if (billing && $('ord-bill') && !$('ord-bill').value) $('ord-bill').value = billing;
  });
  setupAutocomplete('ord-bill', uniqueBillingNames);
  setupAutocomplete('bord-cust', uniqueCustomerNames, name => {
    const billing = billingForCustomer(name);
    if (billing && $('bord-bill') && !$('bord-bill').value) $('bord-bill').value = billing;
  });
  setupAutocomplete('bord-bill', uniqueBillingNames);
  // ord-filter-cust and inw-cust already have debounced 'input' listeners
  // (see DOMContentLoaded above) that refresh their respective lists — picking from the
  // dropdown dispatches an 'input' event above, so those listeners fire without needing
  // a duplicate onSelect callback here.
  setupAutocomplete('ord-filter-cust', uniqueCustomerNames);
}

async function populateBrandFilteredSkus() {
  const brand = ($('ord-brand') && $('ord-brand').value || '').trim();
  const dl = $('dl-skus-brand');
  if (!dl) return;
  state.brandSkuDesc = {};
  if (!brand) { dl.innerHTML = ''; return; }
  try {
    const rows = (await callGS('api_getBrandSkus', brand)) || [];
    setDatalistHtml(dl, rows.map(r =>
      `<option value="${esc(r.sku)}">${esc(r.description || '')}</option>`).join(''));
    rows.forEach(r => { state.brandSkuDesc[String(r.sku).trim().toUpperCase()] = r.description || ''; });
    if ($('oi-sku')) {
      $('oi-sku').placeholder = rows.length
        ? `Start typing SKU (${rows.length} available in ${brand})…`
        : `No sheet named "${brand}" found — add a sheet with Code + Description columns.`;
    }
  } catch (e) {
    console.error(e);
    dl.innerHTML = '';
  }
}

function bindFileDrop(inputId, labelId, handler) {
  const inp = $(inputId);
  const drop = inp.closest('.file-drop');
  inp.addEventListener('change', e => {
    if (e.target.files[0]) { $(labelId).textContent = e.target.files[0].name; handler(e.target.files[0]); }
  });
  drop.addEventListener('dragover', e => { e.preventDefault(); drop.classList.add('dragover'); });
  drop.addEventListener('dragleave', () => drop.classList.remove('dragover'));
  drop.addEventListener('drop', e => {
    e.preventDefault(); drop.classList.remove('dragover');
    if (e.dataTransfer.files[0]) {
      inp.files = e.dataTransfer.files;
      $(labelId).textContent = e.dataTransfer.files[0].name;
      handler(e.dataTransfer.files[0]);
    }
  });
}
function readWorkbook(file) {
  return new Promise((resolve, reject) => {
    const r = new FileReader();
    r.onload = e => { try { resolve(XLSX.read(e.target.result, { type: 'array' })); } catch (err) { reject(err); } };
    r.onerror = reject;
    r.readAsArrayBuffer(file);
  });
}
function sheetToAoA(ws) { return XLSX.utils.sheet_to_json(ws, { header: 1, defval: null, raw: true }); }
function downloadXlsx(filename, data, sheetName) {
  const ws = XLSX.utils.aoa_to_sheet(data);
  const wb = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(wb, ws, sheetName || 'Sheet1');
  XLSX.writeFile(wb, filename);
}

/* ============================================================

 * ============================================================ */
async function handleOutwardFile(file) {
  try {
    const wb = await readWorkbook(file);
    let rows = null;
    if (wb.SheetNames.includes('Pivot_Table')) {
      const aoa = sheetToAoA(wb.Sheets['Pivot_Table']);
      let headerRow = -1;
      for (let i = 0; i < Math.min(aoa.length, 15); i++) {
        const r = (aoa[i] || []).map(v => String(v || '').toLowerCase());
        if (r.some(v => v.includes('row labels') || v === 'sku' || v.includes('code'))
            && r.some(v => v.includes('total') || v.includes('qty') || v.includes('sum'))) { headerRow = i; break; }
      }
      if (headerRow >= 0) {
        const hdr = aoa[headerRow].map(v => String(v || '').toLowerCase());
        const skuCol = hdr.findIndex(v => v.includes('row labels') || v === 'sku' || v.includes('code'));
        const qtyCol = hdr.findIndex(v => v.includes('total') || v.includes('qty') || v.includes('sum'));
        rows = [];
        for (let i = headerRow + 1; i < aoa.length; i++) {
          const r = aoa[i] || [];
          const sku = r[skuCol]; const qty = r[qtyCol];
          if (sku && qty && !String(sku).toLowerCase().includes('grand total')) {
            rows.push({ sku: String(sku).trim(), qty: Number(qty) || 0 });
          }
        }
      }
    }
    if ((!rows || !rows.length) && wb.SheetNames.includes('Clean_data')) {
      const aoa = sheetToAoA(wb.Sheets['Clean_data']);
      const codeCol = (aoa[0] || []).findIndex(v => String(v || '').toLowerCase().includes('code'));
      const map = new Map();
      for (let i = 1; i < aoa.length; i++) {
        const sku = aoa[i] && aoa[i][codeCol >= 0 ? codeCol : 1];
        if (sku) map.set(String(sku).trim(), (map.get(String(sku).trim()) || 0) + 1);
      }
      rows = Array.from(map.entries()).map(([sku, qty]) => ({ sku, qty }));
    }
    if (!rows || !rows.length) {
      const aoa = sheetToAoA(wb.Sheets[wb.SheetNames[0]]);
      rows = [];
      for (let i = 1; i < aoa.length; i++) {
        const [a, b] = aoa[i] || [];
        if (a && b) rows.push({ sku: String(a).trim(), qty: Number(b) || 0 });
      }
    }
    state.outwardParsed = rows.filter(r => r.sku && r.qty !== 0);
    $('outward-preview').textContent =
      `Parsed ${state.outwardParsed.length} rows (source: ${wb.SheetNames.join(', ')})\n\n` +
      state.outwardParsed.slice(0, 12).map(r => `${r.sku.padEnd(24)} ${r.qty}`).join('\n') +
      (state.outwardParsed.length > 12 ? `\n… +${state.outwardParsed.length - 12} more` : '');
    $('btn-upload-outward').disabled = !state.outwardParsed.length;
  } catch (e) { toast('Parse error: ' + e.message, 'err'); }
}

async function uploadOutward() {
  if (!state.outwardParsed || !state.outwardParsed.length) return;
  const btn = $('btn-upload-outward');
  btn.disabled = true; btn.textContent = 'Uploading…';
  try {
    const res = await callGS('api_uploadOutward', state.outwardParsed, $('outward-date').value);
    toast(`Uploaded ${res.added} rows for ${res.date}.`, 'ok');
    $('outward-preview').textContent = '';
    $('outward-filename').textContent = 'No file selected';
    state.outwardParsed = null;
    $('scan-view-date').value = res.date;
    // Refresh masters (may have new SKUs)
    state.masters = await callGS('api_getMasters');
    populateDatalists();
    loadScannedForDate();
    loadDashboard();
  } catch (e) { toast('Upload failed: ' + e.message, 'err'); }
  btn.textContent = 'Upload & Save'; btn.disabled = false;
}

async function loadScannedForDate() {
  const d = $('scan-view-date').value || todayISO();
  const rows = (await callGS('api_getScannedForDate', d)) || [];
  state.outward = rows;
  $('tbl-outward').querySelector('tbody').innerHTML = rows.map(r => `
    <tr><td class="mono">${esc(r.sku)}</td><td>${esc(r.brand)}</td>
    <td class="muted">${esc(r.description)}</td><td class="num">${fmtNum(r.qty)}</td></tr>
  `).join('') || emptyRow(4, `No scanned data for ${d}.`);
}

async function clearOutwardByDate() {
  const d = $('scan-view-date').value;
  if (!d) return toast('Choose a date.', 'err');
  if (!confirm(`Delete ALL scanned rows for ${d}?`)) return;
  try {
    const r = await callGS('api_clearOutwardByDate', d);
    toast(`Deleted ${r.deleted} rows for ${d}.`, 'ok');
    loadScannedForDate(); loadDashboard();
  } catch (e) { toast('Failed: ' + e.message, 'err'); }
}

/* ============================================================
 * BOOK STOCK (Tally) + COMPARISON
 * ============================================================ */
async function handleTallyFile(file) {
  try {
    const wb = await readWorkbook(file);
    const aoa = sheetToAoA(wb.Sheets[wb.SheetNames[0]]);
    let headerRow = -1;
    for (let i = 0; i < Math.min(aoa.length, 8); i++) {
      const r = (aoa[i] || []).map(v => String(v || '').toLowerCase());
      if (r.some(v => v.includes('outward')) && r.some(v => v.includes('closing'))) { headerRow = i; break; }
    }
    let dataStart = headerRow >= 0 ? headerRow + 1 : 0;
    if (headerRow >= 0) {
      const next = (aoa[headerRow + 1] || []).map(v => String(v || '').toLowerCase());
      if (next.some(v => v.includes('quantity') || v.includes('qty'))) dataStart = headerRow + 2;
    }
    const rows = [];
    for (let i = dataStart; i < aoa.length; i++) {
      const r = aoa[i] || [];
      const raw = r[0];
      if (!raw) continue;
      const rawStr = String(raw).replace(/_x000D_/g, '').replace(/\s+/g, ' ').trim();
      if (!rawStr) continue;
      if (/^(sku|code|item|particulars|closing|outward)/i.test(rawStr)) continue;
      const outwardQty = (r[1] === '' || r[1] == null) ? '' : Number(r[1]);
      const closing    = (r[2] === '' || r[2] == null) ? '' : Number(r[2]);
      if (outwardQty === '' && closing === '') continue;
      rows.push({ raw: rawStr, outwardQty, closing });
    }
    state.tallyParsed = rows;
    $('tally-preview').textContent =
      `Parsed ${rows.length} rows (header row: ${headerRow >= 0 ? headerRow + 1 : 'n/a'})\n\n` +
      'SKU'.padEnd(22) + ' OUT   CLOSE   RAW\n' +
      rows.slice(0, 10).map(r => {
        const sku = r.raw.split(/\s+/)[0];
        return `${sku.padEnd(22)} ${String(r.outwardQty).padStart(4)}  ${String(r.closing).padStart(5)}   ${r.raw.slice(0,50)}`;
      }).join('\n') +
      (rows.length > 10 ? `\n… +${rows.length - 10} more` : '');
    $('btn-upload-tally').disabled = !rows.length;
  } catch (e) { toast('Parse error: ' + e.message, 'err'); }
}

}

/* Mismatch-reason dropdown options (used by both comparison tables) */
const REASON_OPTIONS = [
  'Purchase pending',
  'Sale bill pending',
  'Scanned, dispatch pending',
  'Bills made dispatch pending',
];

function reasonCell(type, sku, currentReason, status) {
  if (status === 'MATCH') return '<span class="muted">—</span>';
  const opts = ['<option value="">— select —</option>']
    .concat(REASON_OPTIONS.map(r => `<option${r === currentReason ? ' selected' : ''}>${esc(r)}</option>`))
    .join('');
  return `<select class="reason-sel" data-type="${type}" data-sku="${esc(sku)}" onchange="saveReason(this)">${opts}</select>`;
}

async function saveReason(sel) {
  const d = $('compare-date').value || todayISO();
  const type = sel.dataset.type;
  const sku = sel.dataset.sku;
  try {
    await callGS('api_setMismatchReason', d, type, sku, sel.value);
    sel.style.borderColor = 'var(--ok)';
    setTimeout(() => sel.style.borderColor = '', 800);
  } catch (e) {
    toast('Failed to save reason: ' + e.message, 'err');
    sel.style.borderColor = 'var(--hot)';
  }
}

      populateDatalists();

function renderOrderForm() {
  const brandOpts = ['<option value="">— Select —</option>'].concat((state.masters.brands || []).map(b => `<option>${esc(b)}</option>`)).join('');
  const otypeOpts = ['<option value="">— Select —</option>'].concat((state.masters.orderTypes || []).map(b => `<option>${esc(b)}</option>`)).join('');
  $('ord-of').innerHTML = otypeOpts;
  $('ord-brand').innerHTML = brandOpts;
  $('bord-of').innerHTML = otypeOpts;
  $('bord-brand').innerHTML = brandOpts;
  $('sum-orderof').innerHTML = ['<option value="">All</option>'].concat((state.masters.orderTypes || []).map(o => `<option>${esc(o)}</option>`)).join('');

  const brandSel = $('ord-brand');
  if (brandSel && !brandSel._wired) {
    brandSel._wired = true;
    brandSel.addEventListener('change', () => {
      populateBrandFilteredSkus();
      // Clear the SKU input so user starts fresh under the new brand
      if ($('oi-sku')) $('oi-sku').value = '';
    });
  }
  populateBrandFilteredSkus();

  const oi = $('oi-sku'); const qty = $('oi-qty');
  if (oi && !oi._wired) {
    oi._wired = true;
    oi.addEventListener('keydown', e => {
      if (e.key === 'Enter') { e.preventDefault(); qty.focus(); qty.select(); }
    });
  }
  if (qty && !qty._wired) {
    qty._wired = true;
    qty.addEventListener('keydown', e => {
      if (e.key === 'Enter') { e.preventDefault(); addPendingItem(); }
    });
  }
  renderPendingItems();
}

function addPendingItem() {
  const skuIn = $('oi-sku'); const qtyIn = $('oi-qty');
  const sku = String(skuIn.value || '').trim().toUpperCase();
  const qty = Number(qtyIn.value) || 0;
  if (!sku) { skuIn.focus(); return toast('Enter SKU first.', 'err'); }
  if (qty <= 0) { qtyIn.focus(); qtyIn.select(); return toast('Enter qty > 0.', 'err'); }
  const desc = (state.brandSkuDesc && state.brandSkuDesc[sku]) || '';
  const existing = state.pendingItems.find(x => x.sku === sku);
  if (existing) { existing.qty += qty; if (!existing.description && desc) existing.description = desc; }
  else state.pendingItems.push({ sku, qty, description: desc });
  skuIn.value = ''; qtyIn.value = 1;
  renderPendingItems();
  skuIn.focus();
}

function renderPendingItems() {
  const tbody = $('tbl-pending') && $('tbl-pending').querySelector('tbody');
  if (!tbody) return;
  $('oi-count').textContent = `(${state.pendingItems.length})`;
  tbody.innerHTML = state.pendingItems.length ? state.pendingItems.map((it, i) => `
    <tr>
      <td class="muted">${i + 1}</td>
      <td class="mono">${esc(it.sku)}</td>
      <td><input type="number" min="1" value="${it.qty}" class="pi-qty" data-idx="${i}" style="width:80px" onfocus="this.select()"></td>
      <td class="muted" style="max-width:260px;overflow:hidden;text-overflow:ellipsis;white-space:nowrap" title="${esc(it.description || '')}">${esc(it.description || '')}</td>
      <td><button class="btn danger tiny" onclick="deletePendingItem(${i})">×</button></td>
    </tr>`).join('') : `<tr><td colspan="5" class="muted" style="padding:14px;text-align:center">No items yet.</td></tr>`;
  tbody.querySelectorAll('.pi-qty').forEach(inp => {
    inp.addEventListener('input', e => {
      state.pendingItems[Number(e.target.dataset.idx)].qty = Number(e.target.value) || 0;
    });
  });
}
function deletePendingItem(i) { state.pendingItems.splice(i, 1); renderPendingItems(); }
function clearPendingItems() {
  if (!state.pendingItems.length) return;
  if (!confirm('Clear all pending items?')) return;
  state.pendingItems = []; renderPendingItems();
}

async function placeOrder() {
  const items = state.pendingItems.filter(x => x.sku && x.qty > 0);
  if (!items.length) return toast('Add at least one item.', 'err');
  const brand = $('ord-brand').value;
  const payload = {
    orderOf: $('ord-of').value,
    customerName: $('ord-cust').value,
    billingName: $('ord-bill').value,
    items: items.map(i => ({ sku: i.sku, qty: i.qty, brand, description: i.description || '' })),
  };
  try {
    const r = await callGS('api_placeOrder', payload);
    toast(`Order ${r.orderId} placed (${r.count} lines). Downloading…`, 'ok');
    downloadXlsx(`${r.orderId}.xlsx`, items.map(i => [i.sku, i.qty]), 'Order');
    state.pendingItems = []; renderPendingItems();
    $('ord-cust').value = ''; $('ord-bill').value = '';
    state.masters = await callGS('api_getMasters');
    populateDatalists();
    loadOrders(); loadDashboard();
  } catch (e) { toast('Failed: ' + e.message, 'err'); }
}

async function handleOrderFile(file) {
  try {
    const wb = await readWorkbook(file);
    const aoa = sheetToAoA(wb.Sheets[wb.SheetNames[0]]);
    const rows = [];
    for (let i = 0; i < aoa.length; i++) {
      const r = aoa[i] || [];
      const [a, b, c] = [r[0], r[1], r[2]];
      if (!a || b == null) continue;
      const sku = String(a).trim();
      const qty = Number(b);
      if (isNaN(qty) || qty <= 0) continue;
      if (/^(sku|code|product)/i.test(sku)) continue;
      const desc = c == null ? '' : String(c).trim();
      rows.push({ sku, qty, description: desc });
    }
    state.orderParsed = rows;
    $('order-preview').textContent = `Parsed ${rows.length} rows.\n\n` +
      rows.slice(0, 12).map(r => `${r.sku.padEnd(20)} ${String(r.qty).padStart(4)}  ${(r.description || '').slice(0,40)}`).join('\n');
    $('btn-upload-order').disabled = !rows.length;
  } catch (e) { toast('Parse error: ' + e.message, 'err'); }
}

    const na = results.filter(r => !r.orderId);
    toast(`${ok} allocated. ${na.length} un-ordered.`, na.length ? 'warn' : 'ok');
    $('inward-preview').textContent += '\n\nResult:\n' + results.map(r =>
      r.orderId ? `${r.sku}: ${r.taken} → ${r.orderId} (${r.status})` : `${r.sku}: NOT ORDERED (${r.leftover})`
    ).join('\n');
    state.inwardParsed = null; $('btn-upload-inward').disabled = true;
    // Reload open lines only if a filter is currently set
    if ($('inw-from').value || $('inw-to').value || $('inw-cust').value || $('inw-sku').value) {
      loadOpenOrderLines();
    }
    loadOrders(); loadDashboard();
  } catch (e) { toast('Failed: ' + e.message, 'err'); }
}

function clearSummaryFilters() {
  ['sum-billing','sum-orderof','sum-status','sum-from','sum-to'].forEach(id => $(id).value = '');
  loadSummary();
}
async function loadSummaryBillingNames() {
  try {
    const names = (await callGS('api_getOrderBillingNames')) || [];
    const sel = $('sum-billing');
    const prevVal = sel.value;
    sel.innerHTML = '<option value="">All Billing Names</option>' +
      names.map(name => `<option value="${esc(name)}"${name === prevVal ? ' selected' : ''}>${esc(name)}</option>`).join('');
    if (prevVal && sel.value !== prevVal) sel.value = '';
  } catch (e) {
    console.error('loadSummaryBillingNames error:', e);
  }
}
async function loadSummary() {
  const filters = {
    billing:  $('sum-billing').value,
    orderOf:  $('sum-orderof').value,
    status:   $('sum-status').value,
    dateFrom: $('sum-from').value,
    dateTo:   $('sum-to').value,
  };
  const res = (await callGS('api_getOrderSummary', filters)) || { rows: [] };
  state.lastSummary = res;
  loadSummarySheet();
}

async function loadSummarySheet() {
  const billing = $('sum-billing').value;
  try {
    const rows = (await callGS('api_getSummaryByCustomer', '', billing)) || [];
    const statusPillSummary = s => {
      if (!s) return '';
      if (s === 'Completely Received') return `<span class="pill ok">${esc(s)}</span>`;
      if (s.startsWith('Partially')) return `<span class="pill warn">${esc(s)}</span>`;
      if (s === 'Received More') return `<span class="pill info">${esc(s)}</span>`;
      return `<span class="pill">${esc(s)}</span>`;
    };
    $('tbl-summary-sheet').querySelector('tbody').innerHTML = rows.map(r => `
      <tr>
        <td>${esc(r.customerName)}</td>
        <td class="mono">${esc(r.sku)}</td>
        <td class="num">${fmtNum(r.qtyOrdered)}</td>
        <td class="num">${fmtNum(r.received)}</td>
        <td>${statusPillSummary(r.status)}</td>
        <td>${esc(r.billingName)}</td>
      </tr>`).join('') || emptyRow(6, billing ? 'No summary data matches the selected filters.' : 'No summary data. Place an order first.');
  } catch (e) {
    console.error('loadSummarySheet error:', e);
  }
}

function downloadSummary() {
  const res = state.lastSummary;
  if (!res || !res.rows || !res.rows.length) return toast('No data. Apply filters first.', 'err');
  const wb = XLSX.utils.book_new();
  const lHeader = ['Customer', 'Billing Name', 'Order ID', 'Date', 'Order Of', 'Brand', 'SKU', 'Ordered', 'Received', 'Pending', 'Status'];
  const lData = [lHeader].concat(res.rows.map(row => [
    row['Customer Name'], row['Billing Name'], row['Order ID'], row.Date, row['Order Of'], row.Brand,
    row.SKU, row['Qty Ordered'], row['Qty Received'],
    (Number(row['Qty Ordered']) || 0) - (Number(row['Qty Received']) || 0), row.Status,
  ]));
  XLSX.utils.book_append_sheet(wb, XLSX.utils.aoa_to_sheet(lData), 'Order Lines');
  const filename = `OrderSummary-${todayISO()}.xlsx`;
  XLSX.writeFile(wb, filename);
  toast(`Downloaded ${filename}`, 'ok');
}

function renderMastersLists() {
  const brandOpts = ['<option value="">— Select —</option>'].concat((state.masters.brands || []).map(b => `<option>${esc(b)}</option>`)).join('');
  $('m-brand').innerHTML = brandOpts;
  $('list-brands').innerHTML   = (state.masters.brands || []).map(b => `<li>${esc(b)}</li>`).join('');
  $('list-otypes').innerHTML   = (state.masters.orderTypes || []).map(b => `<li>${esc(b)}</li>`).join('');
  $('list-customers').innerHTML = (state.masters.customers || []).map(c =>
    `<li>${esc(c['Customer Name'])}${c['Billing Name'] ? ' → ' + esc(c['Billing Name']) : ''}</li>`).join('');
  onMasterSkuPick();
}

function onMasterSkuPick() {
  const val = String($('m-sku-pick').value || '').trim().toUpperCase();
  if (!val) {
    $('m-details').classList.add('hidden');
    $('m-empty').classList.remove('hidden');
    return;
  }
  const s = (state.masters.skus || []).find(x => String(x.SKU).toUpperCase() === val);
  if (s) {
    $('m-empty').classList.add('hidden');
    $('m-details').classList.remove('hidden');
    $('m-sku').value = s.SKU;
    $('m-sku').disabled = true;
    $('m-brand').value = s.Brand || '';
    $('m-desc').value = s.Description || '';
    $('m-msl').value = Number(s.MSL) || 0;
  } else {
    $('m-details').classList.add('hidden');
    $('m-empty').classList.remove('hidden');
    $('m-empty').innerHTML = `SKU <b>${esc(val)}</b> not found. Click <b>+ Add SKU</b> to create it.`;
  }
}

function showAddSkuForm() {
  $('m-empty').classList.add('hidden');
  $('m-details').classList.remove('hidden');
  $('m-sku').value = ''; $('m-sku').disabled = false;
  $('m-brand').value = ''; $('m-desc').value = ''; $('m-msl').value = 0;
  $('m-sku').focus();
}

async function saveSku() {
  const sku = $('m-sku').value.trim();
  if (!sku) return toast('SKU required', 'err');
  try {
    state.masters = await callGS('api_upsertSku', sku, $('m-brand').value, $('m-desc').value, $('m-msl').value);
    populateDatalists();
    toast('Saved.', 'ok');
    $('m-sku-pick').value = sku.toUpperCase();
    onMasterSkuPick();
  } catch (e) { toast('Failed: ' + e.message, 'err'); }
}
async function deleteSkuCurrent() {
  const sku = $('m-sku').value.trim();
  if (!sku || !confirm('Delete ' + sku + '?')) return;
  state.masters = await callGS('api_deleteSku', sku);
  populateDatalists();
  $('m-sku-pick').value = '';
  onMasterSkuPick();
}
async function addBrand() {
  const val = ($('brand-new') && $('brand-new').value.trim()) || ($('ord-brand-new') && $('ord-brand-new').value.trim());
  if (!val) return;
  state.masters = await callGS('api_addBrand', val);
  populateDatalists();
  if ($('brand-new')) $('brand-new').value = '';
  if ($('ord-brand-new')) $('ord-brand-new').value = '';
  renderMastersLists(); renderOrderForm();
  toast('Brand added.', 'ok');
}
async function addOrderType() {

  if ($('ord-of-new')) $('ord-of-new').value = '';

}
async function addCustomerMaster() {
  const name = $('cust-new').value.trim();

}
